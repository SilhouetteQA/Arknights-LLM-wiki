[ 03 ]

Agent Pipeline -- 数据流图

查询路由核心流程：意图识别 → 实体提取 → 复杂度分类 → Simple(4层检索) / Complex(LangGraph ReAct 8轮)

graph TD
    Q["用户问题"] --> Wrap["wrap_user_input\n注入防护包裹"]
    Wrap --> Router

    subgraph "router.py -- route_query()"
        Router{"识别结果"} --> IntentLocal["_infer_intent_local\n7 类意图关键词规则"]
        IntentLocal -->|"命中"| Intent
        IntentLocal -->|"未命中"| IntentLLM["recognize_intent_and_rewrite\nLLM 兜底"]
        IntentLLM --> Intent["意图 + 改写后问题"]

        Intent --> EntityLocal["_extract_entities_local\nidentity_map + operators + chapter_timeline"]
        EntityLocal --> FilterHallucination["章节名幻觉过滤\n不在问题文本中的章节名降为 hints"]
        FilterHallucination --> Classify["classify_complexity_local"]

        Classify -->|"simple"| SimplePath
        Classify -->|"complex"| ComplexPath
    end

    subgraph "simple_search.py"
        SimplePath["search_and_collect"] --> ChId["章节识别\nget_chapter_summary 试探"]
        ChId --> L0["Layer 0: Wiki 精确匹配\nget_entity_page"]
        L0 --> L1["Layer 1: Events 章节感知\nsearch_events 按章过滤"]
        L1 --> L2["Layer 2: FAISS 语义搜索\ncosine similarity ≥0.4"]
        L2 --> L3["Layer 3: Dialogue 兜底\n原始文本搜索"]
        L3 --> Collect["去重合并\nexpansion_hints 仅补充 Wiki"]
        Collect --> BuildSimple["build_answer_prompt\n百科编纂者风格"]
        BuildSimple --> CallLLM1["DeepSeek LLM"]
    end

    subgraph "graph.py -- LangGraph ReAct Agent"
        ComplexPath --> InitState["初始化 AgentState\nquestion + messages"]
        InitState --> CallModel["call_model\nLLM + 8 tool definitions"]
        CallModel -->|"tool_calls"| ToolNode["tool_node\n执行检索工具\n结果追加到 messages"]
        ToolNode -->|"继续 (≤8轮)"| CallModel
        CallModel -->|"no tool_calls"| Synthesize["synthesize_node\nSYNTHESIS_PROMPT\n综合证据生成回答"]
    end

    CallLLM1 --> SSE1["SSE 流式输出"]
    Synthesize --> SSE2["SSE 流式输出"]

    SSE1 --> UI["前端 PRTS 终端"]
    SSE2 --> UI

升级阶段架构 -- W1-W4 运行链路与工程化能力

2026-08 升级 P0 全部完成：Observability / Failure Recovery / MCP Server / Planner。默认路由 ReAct（ARKNIGHTS_AGENT_MODE），Planner 可选；工具经恢复链执行，可切换 MCP 后端；全链路 Langfuse trace。

graph TB
    subgraph "运行链路（complex 路径）"
        SRV["server.py\nSSE + checkpoint 工厂"] --> SEL{选择图\nARKNIGHTS_AGENT_MODE}
        SEL -->|"react（默认）"| REACT["build_agent_graph\ncall_model ⇄ tool_node ≤8 轮"]
        SEL -->|"planner（可选）"| PLAN["build_planner_graph\nplan → execute → synthesize\n崩溃检测 → 自动切 ReAct"]
    end

    subgraph "W2 恢复链（工具/LLM 执行层）"
        EX["execute_with_resilience\ntimeout → retry → breaker → fallback"]
        LLMR["chat_completion 重试\n网络/限流/5xx，4xx 不重试"]
        CK["SqliteSaver checkpoint\nthread_id=sha1(问题)，断点续跑"]
    end

    subgraph "W3 MCP Server（工具双轨）"
